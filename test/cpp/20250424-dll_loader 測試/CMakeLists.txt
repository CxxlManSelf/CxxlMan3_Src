# ========================================================================
# 專案名稱：DllLoadTest - 跨平台動態連結庫載入測試
#
# 專案目的：
#   此專案展示如何在跨平台環境中動態載入 DLL/SO/DYLIB 並調用其中的函數。
#   主要測試以下功能：
#   1. 跨平台的動態連結庫加載 (Windows DLL、Linux SO、macOS DYLIB)
#   2. 使用線程池進行多執行緒計算
#   3. 動態獲取並調用 DLL 中導出的函數
#
# 專案結構：
#   - main/  : 主程序專案，負責載入 DLL 並進行測試
#   - plug/  : 插件專案，編譯為動態連結庫，提供四則運算函數
# ========================================================================

# 指定 CMake 最低版本要求
cmake_minimum_required(VERSION 3.10.0)

# 定義專案名稱、版本和使用的語言
project(DllLoadTest VERSION 0.1.0 LANGUAGES C CXX)

# ========================================================================
# C++ 標準設定
# ========================================================================
# 使用 C++20 標準 (需要支援 std::optional、概念等現代 C++ 特性)
set(CMAKE_CXX_STANDARD 20)
# 強制要求必須支援 C++20，若編譯器不支援則報錯
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# 關閉編譯器特定的擴展功能，確保代碼的可移植性
set(CMAKE_CXX_EXTENSIONS OFF)

# ========================================================================
# 輸出目錄設定
# ========================================================================
# 將所有可執行檔案輸出到 build/bin 目錄
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
# 將所有動態連結庫 (.dll/.so/.dylib) 輸出到 build/bin 目錄
# 這樣主程序就能在同一目錄找到插件 DLL
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
# 將所有靜態連結庫 (.lib/.a) 輸出到 build/lib 目錄
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")

# ========================================================================
# 加入子專案
# ========================================================================
# 加入主程序子專案 - 編譯可執行檔案 MyTest
add_subdirectory(main)
# 加入插件子專案 - 編譯動態連結庫 PlugDll
add_subdirectory(plug)
